cmake_minimum_required(VERSION 3.8)
project(hazard_warnings)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
add_compile_options(-Wall -Wextra -Wpedantic)
add_executable(hazard_warnings_node src/hazard_warnings_node.cpp)
target_compile_features(hazard_warnings_node PUBLIC cxx_std_17)
target_include_directories(hazard_warnings_node PRIVATE include)
ament_target_dependencies(hazard_warnings_node rclcpp sensor_msgs std_msgs)
install(TARGETS hazard_warnings_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(hazard_geometry_test test/geometry_test.cpp)
  target_compile_features(hazard_geometry_test PUBLIC cxx_std_17)
  target_include_directories(hazard_geometry_test PRIVATE include)
  add_test(NAME hazard_geometry COMMAND hazard_geometry_test)
endif()
ament_package()
